/* 预加载下一批图片：用户点"加载更多"前就提前拉取，切换更顺滑 */
function preloadNext(){
  const view = getView();
  view.slice(rendered, rendered + BATCH).forEach(p => { const im = new Image(); im.src = imgOf(p); });
}
function cardHTML(p){
  return `<div class="card">
<img loading="lazy" decoding="async" src="${imgOf(p)}" alt="${nameOf(p.no)}" onclick="openLightbox(${p.no})" onerror="this.onerror=null;this.src='${placeholderSVG(p.no)}'">
<div class="name">${nameOf(p.no)}</div>
<div class="meta">#${p.no}${skuOf(p.no) ? ` · <span class="sku">${t().skuLabel} ${skuOf(p.no)}</span>` : ''}</div>
<a class="inquiry" href="${waLinkFor(p.no)}" target="_blank" rel="noopener">${t().inquiry}</a>
</div>`;
}
function renderBatch(){
  const view = getView();
  if(view.length - rendered <= 0){ loadMoreBtn.style.display = 'none'; return; }
  const limit = rendered === 0 ? FIRST_BATCH : BATCH;
  const slice = view.slice(rendered, rendered + limit);
  const frag = document.createDocumentFragment();
  slice.forEach(p => { const d = document.createElement('div'); d.innerHTML = cardHTML(p).trim(); frag.appendChild(d.firstChild); });
  grid.appendChild(frag);
  rendered += slice.length;
  if(rendered >= view.length) loadMoreBtn.style.display = 'none';
  if(!view.length){
    const d = document.createElement('div');
    d.className = 'no-result'; d.textContent = t().noResult;
    grid.appendChild(d);
  }
}
function resetGrid(){ grid.innerHTML = ''; rendered = 0; renderBatch(); preloadNext(); }

function openLightbox(no){
  const p = products.find(x => x.no === no); if(!p) return;
  lbImg.src = imgOf(p);
  lbImg.alt = nameOf(no);
  const skuTxt = skuOf(no) ? ' · ' + t().skuLabel + ' ' + skuOf(no) : '';
  lbInfo.textContent = `${nameOf(no)} · #${no}${skuTxt}`;

  lbDesc.textContent = DESC[no] || '';
  lbDesc.style.display = DESC[no] ? 'block' : 'none';
  lbInquiry.href = waLinkFor(no);
  lightbox.classList.add('active');
}
